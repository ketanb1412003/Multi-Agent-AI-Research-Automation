const sources = $json.sources || [];

// Extract domain from URL
function getDomain(url) {
  if (!url) return "";

  return url
    .replace(/^https?:\/\//, "")
    .replace(/^www\./, "")
    .split("/")[0]
    .toLowerCase();
}

const analyzeSource = (source) => {
  const url = source.url || "";
  const hostname = getDomain(url);

  let authority = "General Website";
  let confidence = 2;

  // Government
  if (
    hostname.includes(".gov") ||   
    hostname.endsWith(".gov") ||
    hostname.includes("nih.gov") ||
    hostname.includes("cdc.gov")
  ) {
    authority = "Government";
    confidence = 5;
  }

  // Universities
  else if (hostname.includes(".edu") || 
           hostname.endsWith(".edu")) {
    authority = "University";
    confidence = 5;
  }

  // Scientific Journals
  else if (
    hostname.includes("nature.com") ||
    hostname.includes("sciencedirect.com") ||
    hostname.includes("springer.com") ||
    hostname.includes("pubmed") ||
    hostname.includes("ncbi.nlm.nih.gov") ||
    hostname.includes("pmc.ncbi.nlm.nih.gov") ||
    hostname.includes("ieee.org") ||
    hostname.includes("acm.org")
  ) {
    authority = "Scientific Journal";
    confidence = 5;
  }

  // International Organizations
  else if (
    hostname.includes("who.int") ||
    hostname.includes("un.org") ||
    hostname.includes("unesco.org")
  ) {
    authority = "International Organization";
    confidence = 5;
  }

  // Official Companies
  else if (
    hostname.includes("google.com") ||
    hostname.includes("deepmind.com") ||
    hostname.includes("openai.com") ||
    hostname.includes("microsoft.com") ||
    hostname.includes("ibm.com") ||
    hostname.includes("nvidia.com")
  ) {
    authority = "Official Company";
    confidence = 4;
  }

  // News
  else if (
    hostname.includes("bbc.com") ||
    hostname.includes("cnn.com") ||
    hostname.includes("reuters.com") ||
    hostname.includes("apnews.com")
  ) {
    authority = "News";
    confidence = 3;
  }

  // Blogs
  else if (
    hostname.includes("medium.com") ||
    hostname.includes("dev.to") ||
    hostname.includes("substack.com")
  ) {
    authority = "Blog";
    confidence = 2;
  }

  return {
    ...source,
    domain: hostname,
    authority,
    confidence
  };
};

return {
  json: {
    research_task: $json.research_task,
    total_sources: sources.length,
    sources: sources.map(analyzeSource)
  }
};
